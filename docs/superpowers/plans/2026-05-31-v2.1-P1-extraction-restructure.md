> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 s01b 一次性巨型 prompt 提取重构为分阶段（S1 相关性门→S4 实体→S2 基本信息→S3 金融→S5 摘要）、各自聚焦+校验的提取链，去掉 crewAI 薄壳，并清理 guardrail 硬编码枚举，系统性降低提取错误率（以 P0 字段级准确率度量）。

```python
"""配置化统一 LLM client：一处配 provider+model+key，各阶段经 complete() 调用；处理重试/退避/JSON 容错。"""
import json
import logging
import os
import time
from dataclasses import dataclass, field

logger = logging.getLogger(__name__)


@dataclass
class LLMConfig:
    """LLM 调用配置；provider∈{gemini,deepseek}，可被各阶段单独覆盖 model。"""
    provider: str = "gemini"
    model: str = "gemini-2.5-flash-lite"
    temperature: float = 0.14
    max_retries: int = 3
    backoff_base: float = 2.0
    base_url: str = ""        # deepseek 用（OpenAI 兼容）
    _api_key: str = field(default="", repr=False)


def config_from_env(stage: str | None = None) -> LLMConfig:
    """从环境变量构建配置：S01B_PROVIDER / S01B_MODEL，缺省 gemini-2.5-flash-lite。stage 预留按阶段覆盖。"""
    provider = os.getenv("S01B_PROVIDER", "gemini").strip().lower()
    if provider == "deepseek":
        return LLMConfig(
            provider="deepseek",
            model=os.getenv("DEEPSEEK_MODEL", "").strip(),
            base_url=os.getenv("DEEPSEEK_BASE_URL", "https://api.deepseek.com").strip(),
            _api_key=os.getenv("DEEPSEEK_API_KEY", "").strip(),
        )
    return LLMConfig(
        provider="gemini",
        model=os.getenv("S01B_MODEL", "gemini-2.5-flash-lite").strip(),
        _api_key=os.getenv("GEMINI_API_KEY", "").strip(),
    )


def _build_provider_client(cfg: LLMConfig):
    """按 provider 构建底层 SDK client；缺 key 立即抛错（rule#4 入口校验）。"""
    if not cfg._api_key:
        raise RuntimeError(f"LLM provider={cfg.provider} 缺少 API key（请检查 .env）")
    if cfg.provider == "deepseek":
        from openai import OpenAI
        return OpenAI(api_key=cfg._api_key, base_url=cfg.base_url or None)
    from google import genai
    return genai.Client(api_key=cfg._api_key)


class LLMClient:
    """统一调用入口：complete(prompt, *, stage, json_mode) -> dict|str；按 provider 分派 + 重试退避。"""

    def __init__(self, cfg: LLMConfig, _client=None):
        self.cfg = cfg
        self._client = _client if _client is not None else _build_provider_client(cfg)

    def _raw_call(self, prompt: str, json_mode: bool) -> str:
        """单次底层调用，返回响应文本。"""
        if self.cfg.provider == "deepseek":
            resp = self._client.chat.completions.create(
                model=self.cfg.model,
                messages=[{"role": "user", "content": prompt}],
                temperature=self.cfg.temperature,
                response_format={"type": "json_object"} if json_mode else None,
            )
            return resp.choices[0].message.content
        from google.genai import types
        cfg_kwargs = {"temperature": self.cfg.temperature}
        if json_mode:
            cfg_kwargs["response_mime_type"] = "application/json"
        resp = self._client.models.generate_content(
            model=self.cfg.model, contents=prompt,
            config=types.GenerateContentConfig(**cfg_kwargs),
        )
        return resp.text

    def complete(self, prompt: str, *, stage: str, json_mode: bool = True):
        """跑一次补全：失败按 max_retries 指数退避；json_mode 时解析为 dict；全失败抛 RuntimeError（不静默）。"""
        last_error = ""
        for attempt in range(1, self.cfg.max_retries + 1):
            try:
                text = self._raw_call(prompt, json_mode)
                return json.loads(text) if json_mode else text
            except Exception as e:  # 兜底重试；具体上下文写 logging（rule#1）
                last_error = f"{type(e).__name__}: {e}"
                logger.error("LLM complete 失败 stage=%s attempt=%d/%d error=%s",
                             stage, attempt, self.cfg.max_retries, last_error)
                if attempt < self.cfg.max_retries and self.cfg.backoff_base:
                    time.sleep(self.cfg.backoff_base ** attempt)
        raise RuntimeError(f"LLM complete stage={stage} 失败(重试{self.cfg.max_retries}次): {last_error}")
```

```python
"""s01b 知识表加载器：主要设备清单（A.5）与容量系数（A.3）；从本包内 JSON 读取，lru_cache。"""
import json
import pathlib
from functools import lru_cache

_DIR = pathlib.Path(__file__).resolve().parent


@lru_cache(maxsize=1)
def _equipment() -> dict:
    """读 main_equipment.json。"""
    return json.loads((_DIR / "main_equipment.json").read_text(encoding="utf-8"))


@lru_cache(maxsize=1)
def _factors() -> dict:
    """读 capacity_factors.json（含 _note 注释键）。"""
    return json.loads((_DIR / "capacity_factors.json").read_text(encoding="utf-8"))


def main_equipment(category_zh: str) -> list[str]:
    """某项目类型的主要设备清单；未知类型返回空表。"""
    return list(_equipment().get(category_zh, []))


def capacity_factor(category_zh: str) -> float | None:
    """某项目类型的容量系数；未知返回 None。"""
    v = _factors().get(category_zh)
    return float(v) if isinstance(v, (int, float)) else None


def derive_capacity_mw(annual_generation_mwh: float, category_zh: str) -> float | None:
    """A.3：由年发电量(MWh)反推装机容量(MW)= gen/(8760*factor)；无系数返回 None。"""
    f = capacity_factor(category_zh)
    if not f or annual_generation_mwh <= 0:
        return None
    return round(annual_generation_mwh / (8760.0 * f), 1)
```

```python
def build_s1_prompt(title: str, content: str, date: str) -> str:
    """构建 S1 门控 prompt：输出 {status, reasoning, project_subject, is_power_generation_plant}。"""
    return f"""你是一个能源新闻相关性审核专家。只做一件事：判断这篇新闻是否是「有中国企业参与的海外可再生能源/电力电站项目」，并给出三选一结论。**不要提取项目明细。**

- "valid_project"：确有具体的海外电站/电力工程项目，且有中国企业参与（业主/承包商/设备商/投资方/融资方等任一）。
- "no_chinese_participation"：是可再生能源/电力项目，但正文中无任何中国企业出现，或各环节参与者均非中企。
- "irrelevant"：不属于我们要收录的「中企参与的海外发电/电力工程项目」。

### 判 irrelevant 的强制排除规则（命中任一即 irrelevant）
1. **非发电主体的基建/民用工程**：学校、医院、机场、铁路、港口、农业灌溉、供水管网、工业园区、住宅等——即便由中企 EPC、即便提到「电力配套/供电工程/屋顶光伏」但仅作附属设施且无明确装机容量。原则：要的是「发电厂」不是「用电户」。
   - **数据中心特例（A.2）**：数据中心只有在「由可再生能源专门供电/自建配套 RE 电站」时才算 valid；普通市电供电的数据中心 → irrelevant。
2. **油气/矿业/管网**：油田、气田、炼化、LNG、油气管道、采矿、冶炼等非发电项目，即便中企参与 → irrelevant。
```

```python
def _run_stage_with_retry(run_fn, validate_fn, *, stage: str, retries: int = 2) -> dict:
    """跑某阶段并校验：校验不过则重试；仍不过记 logging.error 并返回最后一次输出（不静默吞，交由上层/批级 guardrail 兜底）。"""
    last = None
    for attempt in range(1, retries + 1):
        last = run_fn()
        errors = validate_fn(last)
        if not errors:
            return last
        logger.error("阶段 %s 校验未过 attempt=%d/%d errors=%s", stage, attempt, retries, errors)
    return last if last is not None else {}
```

```python
"""母子公司/曾用名归一：文中实体名 → (顶级集团规范名, 子公司规范名)；复用 company_hierarchy.json。"""
import logging

logger = logging.getLogger(__name__)


def _hierarchy_index() -> dict:
    """构建「任意别名/子公司名 → (顶级集团, 子公司规范名)」索引，复用 tool 的 hierarchy 数据。"""
    from s01b_project_extractor.tools.project_extractor_tool import _load_hierarchy_data
    data = _load_hierarchy_data()  # 现有缓存加载器
    index = {}
    # data 结构以实际 company_hierarchy.json 为准：顶级集团 → {aliases, subsidiaries:[{name, aliases}]}
    for parent, meta in (data or {}).items():
        index[parent] = (parent, "")
        for alias in (meta.get("aliases") or []):
            index[alias] = (parent, "")
        for sub in (meta.get("subsidiaries") or []):
            sub_name = sub.get("name") if isinstance(sub, dict) else sub
            if sub_name:
                index[sub_name] = (parent, sub_name)
                for sa in (sub.get("aliases") or [] if isinstance(sub, dict) else []):
                    index[sa] = (parent, sub_name)
    return index


def normalize_entity(name: str) -> tuple[str, str]:
    """实体名归一为 (顶级集团, 子公司)；空→('Unknown','')；未匹配→原名透传（子公司空）。"""
    name = (name or "").strip()
    if not name:
        return "Unknown", ""
    idx = _hierarchy_index()
    if name in idx:
        return idx[name]
    # 包含匹配兜底（文中名含规范名/规范名含文中名）
    for key, (parent, sub) in idx.items():
        if key and (key in name or name in key):
            return parent, sub
    return name, ""
```
> **执行注意**：先读 `company_hierarchy.json` 确认其真实结构（键是否为顶级集团、是否有 `aliases`/`subsidiaries` 字段、`_get_entity_lookup_text` 如何解析它）。本函数的解析逻辑须与实际 JSON 结构对齐——若结构不同，按实际结构改 `_hierarchy_index`，保持「别名/子公司 → 顶级集团」语义不变。

```python
"""S4 阶段：抽取某项目的中方实体并归一母子公司，校验 involvement_type 合法。"""
import bri_enums as enums
from s01b_project_extractor.stages.entity_normalize import normalize_entity


def run_s4(title: str, content: str, project_name: str, category: str, *, llm) -> list[dict]:
    """跑 S4 抽取并对每个实体做母子公司归一；返回 chinese_entities 列表。"""
    from s01b_project_extractor.prompts.s4_entity import build_s4_prompt
    from s01b_project_extractor.tools.project_extractor_tool import _get_entity_lookup_text
    from s01b_project_extractor.knowledge import main_equipment
    prompt = build_s4_prompt(title, content, project_name, category,
                             _get_entity_lookup_text(), main_equipment(category))
    out = llm.complete(prompt, stage="s4", json_mode=True)
    ents = out.get("chinese_entities", []) if isinstance(out, dict) else []
    for e in ents:
        raw = e.get("Implementing Chinese Entity", "")
        parent, sub = normalize_entity(raw)
        e["Implementing Chinese Entity"] = parent
        # 若 S4 自己给了 Subsidiary 就保留，否则用归一得到的
        if not (e.get("Subsidiary Name") or "").strip() and sub:
            e["Subsidiary Name"] = sub
    return ents


def validate_s4(entities: list[dict]) -> list[str]:
    """校验：每个实体 involvement_type 合法、Implementing Chinese Entity 非空。"""
    errors = []
    valid_inv = set(enums.valid_involvement())
    for i, e in enumerate(entities):
        inv = str(e.get("Type of involvement of Chinese Entities", "")).strip()
        if inv and inv not in valid_inv:
            errors.append(f"实体#{i} 参与方式不合法: {inv}")
        if not str(e.get("Implementing Chinese Entity", "")).strip():
            errors.append(f"实体#{i} Implementing Chinese Entity 为空")
    return errors
```

```python
### 项目识别边界（防过度拆分，最重要）
- **一个物理电站/工程 = 一个项目**，即使新闻提到它的多个阶段（签约/开工/投产）或多方参与，也只算**一条**。
- 同一新闻同一项目的「一期/二期」若为**同一规划下的分期**，仍可合并为一条并在名称注明分期；只有明确是**独立选址/独立立项**的不同电站才拆成多条。
- 「框架协议/总量供货/国家级目标」不指向具体电站的，**不要**编成项目（这类由相关性步骤已大多剔除）。
- 不确定是否同一项目时，**倾向合并**而非拆分。

### 关键规则
- **A.1 分类映射**：垃圾发电/生物质发电→「生物质」；光热→按新能源发电归「光伏」或「综合能源」（若与光伏并存）；组件厂/锂电厂/多晶硅厂/装备产业园→「新能源制造业」（看产出单位：GW/年、万吨/年=工厂；MW/GW 装机=电站）；仅输电线路/变电站/电网改造/升压站→「输变电建设」；火电/燃机/燃气电厂/煤电（不与新能源结合）→「火电」。
- **混合能源（光伏+储能等）**：Category=「综合能源」；Capacity (MW)=各子项总装机数字之和；Capacity Details 填明细如「PV: 100MW, BESS: 50MW」。
- **A.3 容量反推**：若正文只有年发电量（如「年发电 X 亿千瓦时」）而无装机容量，按容量系数反推装机并在 Capacity Details 注明「capacity_source=derived」。
- **B.5 阶段细化**：「签临时移交证书/PAC」=「竣工」非「投入运营」；「项目公司与咨询公司签约」≠项目本身签约（不要据此判签约）；「并网发电/投产/商业运行」=「投入运营」；「完工」=「竣工」；仅「投标/招标/竞标/投标意向」且无中企具体中标/签约=「拟开发」。
```

```python
"""S2 阶段：项目识别+基本信息；in-memory 校验 category/stage 枚举与名称含中文。"""
import re
import bri_enums as enums

_CHINESE_RE = re.compile(r"[一-鿿]")


def run_s2(title: str, content: str, date: str, *, llm) -> list[dict]:
    """跑 S2，返回 projects 列表（每项含 common_info）。"""
    from s01b_project_extractor.prompts.s2_basic import build_s2_prompt
    out = llm.complete(build_s2_prompt(title, content, date), stage="s2", json_mode=True)
    return out.get("projects", []) if isinstance(out, dict) else []


def validate_s2(projects: list[dict]) -> list[str]:
    """校验每个项目：category/stage 合法、Project Name 含中文。"""
    errors = []
    valid_cat = set(enums.valid_categories())
    valid_stage = set(enums.valid_stages())
    for i, p in enumerate(projects):
        ci = p.get("common_info", {}) or {}
        cat = str(ci.get("Category", "")).strip()
        if cat and cat not in valid_cat:
            errors.append(f"项目#{i} 分类不合法: {cat}")
        stage = str(ci.get("Project Stage", "")).strip()
        if stage and stage not in valid_stage:
            errors.append(f"项目#{i} 阶段不合法: {stage}")
        pname = str(ci.get("Project Name", "")).strip()
        if pname and not _CHINESE_RE.search(pname):
            errors.append(f"项目#{i} 名称未含中文: {pname[:30]}")
    return errors
```

```python
"""S5 阶段：生成摘要并按高置信度纠偏 common_info 的 stage/category。"""
import bri_enums as enums


def run_s5(structured_project: dict, *, llm) -> dict:
    """跑 S5，返回 {project_summary, corrections}。"""
    from s01b_project_extractor.prompts.s5_summary import build_s5_prompt
    out = llm.complete(build_s5_prompt(structured_project), stage="s5", json_mode=True)
    return out if isinstance(out, dict) else {}


def apply_s5(project: dict, s5_out: dict) -> dict:
    """把 S5 摘要写入 project；高置信度(≥0.8)且枚举合法的纠偏应用到 common_info。"""
    summary = (s5_out.get("project_summary") or "").strip()
    if summary:
        project["project_summary"] = summary
    ci = project.setdefault("common_info", {})
    valid = {"project_stage": set(enums.valid_stages()), "category": set(enums.valid_categories())}
    field_map = {"project_stage": "Project Stage", "category": "Category"}
    for c in (s5_out.get("corrections") or []):
        f = c.get("field"); sv = str(c.get("suggested_value", "")).strip()
        if f in field_map and float(c.get("confidence", 0)) >= 0.8 and sv in valid.get(f, set()):
            ci[field_map[f]] = sv
    return project
```
