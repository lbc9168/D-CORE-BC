```tsx
/** 年份胶囊选择器（cobalt 风格）：一排圆角胶囊，选中者 brand 实底；首位"全部" */
export default function YearPills({
  years,
  value,
  onChange,
  allLabel,
}: {
  years: number[];
  value: number | null; // null = 全部年份
  onChange: (v: number | null) => void;
  allLabel: string;
}) {
  const pill = (active: boolean) =>
    `px-3 py-1 rounded-full text-xs font-semibold transition-colors select-none ${
      active ? "bg-brand-600 text-white shadow-sm" : "text-slate-500 hover:bg-slate-100"
    }`;
  return (
    <div className="flex items-center gap-0.5 bg-white border border-slate-200 rounded-full p-1 shadow-sm">
      <button className={pill(value == null)} onClick={() => onChange(null)}>
        {allLabel}
      </button>
      {years.map((y) => (
        <button key={y} className={pill(value === y)} onClick={() => onChange(y)}>
          {y}
        </button>
      ))}
    </div>
  );
}
```

```ts
import type { Project, VizGroup } from "@/types";
import { statusGroupAt } from "@/contexts/DataContext";
import { hasCapacity } from "@/utils/capacityPolicy";

export interface YearStat {
  year: number;
  active: number;      // 活跃项目数（year ∈ [first,last]）
  capacityMw: number;  // 活跃且 hasCapacity 项目的装机合计
  countries: number;   // 活跃项目覆盖国家数
  entities: number;    // 活跃项目 distinct primary_entity 数
}

/** 选定年（null=全部）下的活跃项目集合 */
export function activeProjectsAt(projects: Project[], year: number | null): Project[] {
  if (year == null) return projects;
  return projects.filter(
    (p) =>
      p.first_event_year != null &&
      p.last_event_year != null &&
      p.first_event_year <= year &&
      year <= p.last_event_year,
  );
}

/** 每个可用年份的四项 KPI 序列（喂 KPI 卡迷你趋势线） */
export function buildYearSeries(projects: Project[], years: number[]): YearStat[] {
  return years.map((year) => {
    const active = activeProjectsAt(projects, year);
    let capacityMw = 0;
    const countries = new Set<string>();
    const entities = new Set<string>();
    for (const p of active) {
      if (hasCapacity(p.category)) capacityMw += p.capacity_mw ?? 0;
      if (p.country_iso) countries.add(p.country_iso);
      if (p.primary_entity) entities.add(p.primary_entity);
    }
    return { year, active: active.length, capacityMw, countries: countries.size, entities: entities.size };
  });
}

/** 选定年下 4 大类状态分布（year=null 用项目最终状态） */
export function statusMix(projects: Project[], year: number | null): Map<VizGroup, number> {
  const m = new Map<VizGroup, number>();
  for (const p of projects) {
    const g = statusGroupAt(p, year);
    if (g) m.set(g, (m.get(g) ?? 0) + 1);
  }
  return m;
}

/** Top N 国家（按项目数或装机容量） */
export function topCountries(
  projects: Project[],
  metric: "count" | "capacity",
  n = 12,
): { iso: string; value: number }[] {
  const m = new Map<string, number>();
  for (const p of projects) {
    if (!p.country_iso) continue;
    if (metric === "count") {
      m.set(p.country_iso, (m.get(p.country_iso) ?? 0) + 1);
    } else {
      if (!hasCapacity(p.category)) continue;
      m.set(p.country_iso, (m.get(p.country_iso) ?? 0) + (p.capacity_mw ?? 0));
    }
  }
  return Array.from(m, ([iso, value]) => ({ iso, value }))
    .sort((a, b) => b.value - a.value)
    .slice(0, n);
}

/** 大数字格式化：1234 → 1,234；123456 → 123.5K；1234567 → 1.2M */
export function formatBig(v: number): string {
  if (v >= 1_000_000) return (v / 1_000_000).toFixed(1) + "M";
  if (v >= 100_000) return (v / 1_000).toFixed(1) + "K";
  return Math.round(v).toLocaleString("en-US");
}
```

```tsx
import { useMemo } from "react";
import { PieChart, Pie, Cell, ResponsiveContainer } from "recharts";
import type { Project, VizGroup } from "@/types";
import { VIZ_GROUPS, VIZ_GROUP_KEYS } from "@/types";
import { statusMix } from "@/utils/overviewStats";
import { useLang } from "@/contexts/LanguageContext";

/** 4 大类状态构成：左 donut + 右占比表 */
export default function StatusDonut({
  projects,
  year,
  title,
  shareLabel,
}: {
  projects: Project[];

  title: string;
  shareLabel: string;

                   paddingAngle={2} isAnimationActive={false}>
                {rows.map((r) => <Cell key={r.key} fill={r.color} stroke="none" />)}

            </PieChart>
          </ResponsiveContainer>
        </div>

          <thead>
```

```tsx
import { useEffect, useRef } from "react";
import type { Project } from "@/types";
import ProjectDetail from "./ProjectDetail";

/**
 * 项目详情滑出抽屉：固定在视口右缘的覆盖图层（不挤占页面布局），
 * 半透明遮罩点击或 Esc 关闭；project=null 时滑出收起。
 */
export default function ProjectDetailDrawer({
  project,
  onClose,
}: {
  project: Project | null;
  onClose: () => void;
}) {
  const open = project != null;
  // 关闭动画期间保留最后内容，避免滑出途中内容闪空
  const lastRef = useRef<Project | null>(null);
  if (project) lastRef.current = project;
  const shown = project ?? lastRef.current;

  useEffect(() => {
    if (!open) return;
    const onKey = (e: KeyboardEvent) => {
      if (e.key === "Escape") onClose();
    };
    document.addEventListener("keydown", onKey);
    return () => document.removeEventListener("keydown", onKey);
  }, [open, onClose]);

  return (
    <>
      {/* 遮罩 */}
      <div
        className={`fixed inset-0 z-40 bg-slate-900/20 transition-opacity duration-300 ${
          open ? "opacity-100" : "opacity-0 pointer-events-none"
        }`}
        onClick={onClose}
        aria-hidden
      />
      {/* 抽屉面板 */}
      <aside
        className={`fixed inset-y-0 right-0 z-50 w-[26rem] max-w-[92vw] bg-white shadow-2xl
                    border-l border-slate-200 transition-transform duration-300 ease-in-out ${
                      open ? "translate-x-0" : "translate-x-full"
                    }`}
        role="dialog"
        aria-modal="true"
      >
        {shown && <ProjectDetail project={shown} onClose={onClose} />}
      </aside>
    </>
  );
}
```

## 设计决策备忘（执行中如有冲突按此裁决）

1. **Overview 与 FilterBar 解耦**：Overview 只受自身年份胶囊控制、消费 `allProjects` 全集——这是有意为之（仪表盘呈现"全库画像"，与地图页的筛选探索互不干扰）。不要把 FilterBar 装进 Overview。
2. **"活跃项目"口径** = `first_event_year ≤ Y ≤ last_event_year`，与 FilterBar `refYear` 的区间存在性语义一致；容量合计始终用 `hasCapacity()` 排除储能/输变电。该口径已写进页面说明条，改口径必须同步改说明文案。
3. **Drawer 是覆盖层不是布局列**：`position: fixed` + translate 动画 + 遮罩，绝不挤压地图/表格宽度（用户明确要求）。
4. **配色不动**：brand 深青绿体系 + 既有 category/stage/viz_group 色板；cobalt 只借版式不借色。
5. **未来"企业信息"页**：只需 ① `Sidebar.tsx` 的 `NAV_ITEMS` 加一行；② `App.tsx` 加一条 Route；③ i18n 加键。本次不实现。
6. **数据是 5 月底旧导出**：视觉验证可用，但不要据此调整任何统计口径或"哪个类别值得显示"的判断；新数据等主线 s02a 重建→s03 后自动替换。
