```python
"""review_queue：E.1/E.2 统一复核清单（review.db 新表 review_queue）。"""
import logging
import sqlite3

logger = logging.getLogger(__name__)

_REASON_TYPES = ("unnamed_entity", "low_confidence_field", "status_challenge")


def init_queue(db_path) -> None:
    """建 review_queue 表（幂等）。"""
    conn = sqlite3.connect(str(db_path))
    try:
        conn.execute(
            """CREATE TABLE IF NOT EXISTS review_queue (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                canonical_id TEXT, source_link TEXT,
                reason_type TEXT NOT NULL, field TEXT,
                current_value TEXT, suggested_value TEXT,
                critic_confidence REAL, evidence TEXT,
                state TEXT NOT NULL DEFAULT 'pending', created_at TEXT
            )"""
        )
        conn.commit()
    finally:
        conn.close()


def add_item(db_path, *, canonical_id, source_link, reason_type, field, current_value,
             suggested_value, critic_confidence, evidence) -> None:
    """插入一条复核项；reason_type 必须合法，否则抛错（入口校验）。"""
    if reason_type not in _REASON_TYPES:
        raise ValueError(f"非法 reason_type={reason_type!r}，合法={_REASON_TYPES}")
    conn = sqlite3.connect(str(db_path))
    try:
        conn.execute(
            """INSERT INTO review_queue
               (canonical_id, source_link, reason_type, field, current_value,
                suggested_value, critic_confidence, evidence, state, created_at)
               VALUES (?,?,?,?,?,?,?,?, 'pending', datetime('now'))""",
            (canonical_id, source_link, reason_type, field, current_value,
             suggested_value, critic_confidence, evidence),
        )
        conn.commit()
    finally:
        conn.close()


def pending(db_path) -> list[dict]:
    """读所有 state='pending' 的复核项为 dict 列表。"""
    conn = sqlite3.connect(str(db_path))
    conn.row_factory = sqlite3.Row
    try:
        rows = [dict(r) for r in conn.execute(
            "SELECT * FROM review_queue WHERE state='pending' ORDER BY id")]
    finally:
        conn.close()
    return rows
```

```python
"""signals：确定性预筛——对单个项目记录算可疑字段，挑出值得花一次批判调用的候选。"""
import sys
import pathlib

_root = next(p for p in pathlib.Path(__file__).resolve().parents if (p / "bri_enums.py").exists())
if str(_root) not in sys.path:
    sys.path.insert(0, str(_root))
import bri_enums  # noqa: E402

_PLACEHOLDER = "未具名中企"


def _is_unknown(v) -> bool:
    """值是否为空/Unknown。"""
    s = str(v or "").strip()
    return s == "" or s.lower() == "unknown"


def suspicious_fields(proj: dict) -> list[str]:
    """返回该项目的可疑字段名列表（空=不可疑）。proj 用 eval.predict.flat_prediction 的扁平结构。"""
    out: list[str] = []
    cat = proj.get("category")
    ents = proj.get("entities") or []
    real_ents = [e for e in ents if not e.get("is_unnamed_placeholder")
                 and str(e.get("name", "")).strip() != _PLACEHOLDER]
    # 实体：有真实实体但名字 Unknown / 或一个真实实体都没有（且非 unnamed 占位场景）
    if ents and all(_is_unknown(e.get("name")) for e in real_ents) and real_ents:
        out.append("entities")
    elif not ents:
        out.append("entities")
    # 容量：该类目本应有容量却空
    if cat and not bri_enums.is_no_capacity(cat) and _is_unknown(proj.get("capacity_mw")):
        out.append("capacity_mw")
    # 参与方式：Unknown/空
    if _is_unknown(proj.get("involvement_type")):
        out.append("involvement_type")
    return out
```

```python
"""critic：批判 agent——重读原文质疑某项目的提取字段并挑战相关性，输出经 pydantic 校验的 verdict 列表。"""
import json
import logging
from typing import Literal, Optional

from pydantic import BaseModel, Field, ValidationError

logger = logging.getLogger(__name__)


class Verdict(BaseModel):
    """单字段核验结论：verdict 三选一，confidence∈[0,1]。"""
    field: str
    verdict: Literal["ok", "wrong", "unsupported"]
    suggested_value: Optional[str] = None
    confidence: float = Field(ge=0.0, le=1.0)
    evidence: str = ""


_PROMPT = """你是严谨的核查员。下面是一篇新闻原文和从中提取的某个项目记录。
请逐字段核验提取是否被原文支持，并额外挑战：这真的是"有中国企业参与的可再生能源项目"吗（还是油气/矿业/民用工程被误收）？

原文标题：{title}
原文正文：{content}

提取记录（JSON）：{project}
重点核查这些可疑字段：{suspicious}

只输出 JSON：{{"verdicts":[{{"field":"字段名或 status","verdict":"ok|wrong|unsupported","suggested_value":"建议值或null","confidence":0.0,"evidence":"原文依据"}}]}}
- verdict=ok 表示原文支持当前值；wrong 表示错且给 suggested_value；unsupported 表示原文找不到依据。
- 对"是否相关/是否真有中企项目"的判断，field 写 "status"。"""


def review_project(*, article: dict, project: dict, suspicious: list, llm) -> list:
    """对一个可疑项目跑批判核验；返回 Verdict 列表；LLM/解析失败由 llm 层抛出，输出非法则抛 ValidationError。"""
    prompt = _PROMPT.format(
        title=article.get("title", ""), content=article.get("content", ""),
        project=json.dumps(project, ensure_ascii=False), suspicious=suspicious,
    )
    raw = llm.complete(prompt, stage="critic", json_mode=True)
    items = raw.get("verdicts", []) if isinstance(raw, dict) else []
    try:
        return [Verdict.model_validate(it) for it in items]
    except ValidationError as e:
        logger.error("critic 输出校验失败: %s | 原始=%s", e, raw)
        raise
```

```python
"""apply：按 verdict 分流——字段值高置信纠正自动改库；status/存在性挑战与低置信入 review_queue（不自动删）。"""
import logging
from review import queue

logger = logging.getLogger(__name__)


def route(*, canonical_id, source_link, verdicts, review_db, update_field, conf_threshold: float = 0.8) -> dict:
    """处理一个项目的 verdict 列表；update_field(canonical_id, field, new_value) 由调用方注入（事务改库）。
    返回 {auto_corrected, queued}。"""
    auto_corrected = queued = 0
    for v in verdicts:
        if v.verdict == "ok":
            continue
        # status / 存在性挑战：高风险，绝不自动改，一律入清单
        if v.field == "status":
            queue.add_item(review_db, canonical_id=canonical_id, source_link=source_link,
                           reason_type="status_challenge", field="status",
                           current_value=None, suggested_value=v.suggested_value,
                           critic_confidence=v.confidence, evidence=v.evidence)
            queued += 1
            continue
        # 字段值纠正：高置信 + 有建议值 → 自动改；否则入清单
        if v.verdict == "wrong" and v.suggested_value and v.confidence >= conf_threshold:
            update_field(canonical_id, v.field, v.suggested_value)
            auto_corrected += 1
        else:
            queue.add_item(review_db, canonical_id=canonical_id, source_link=source_link,
                           reason_type="low_confidence_field", field=v.field,
                           current_value=None, suggested_value=v.suggested_value,
                           critic_confidence=v.confidence, evidence=v.evidence)
            queued += 1
    return {"auto_corrected": auto_corrected, "queued": queued}
```
