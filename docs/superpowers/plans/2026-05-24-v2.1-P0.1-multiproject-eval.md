```python
import re

NAME_THRESHOLD = 80  # rapidfuzz 项目名相似度阈值（0-100）
_WS = re.compile(r"\s+")

def _norm_name(s) -> str:
    """项目/实体名归一：去空白、转小写，供相似度与集合比对。"""
    return _WS.sub("", str(s or "").strip().lower())

def _name_sim(a, b) -> float:
    """两个项目名的 rapidfuzz token_set_ratio（归一后；任一为空返回 0）。"""
    from rapidfuzz import fuzz
    na, nb = _norm_name(a), _norm_name(b)
    if not na or not nb:
        return 0.0
    return fuzz.token_set_ratio(na, nb)

def _attr_bonus(p: dict, g: dict) -> int:
    """对齐平局打破的属性加分：同 category +1、capacity_mw 在 ±5% 内 +1。"""
    bonus = 0
    if p.get("category") and p.get("category") == g.get("category"):
        bonus += 1
    if cmp_numeric(p.get("capacity_mw"), g.get("capacity_mw")):
        bonus += 1
    return bonus

def align_projects(pred_projects, gold_projects, threshold: int = NAME_THRESHOLD):
    """贪心对齐预测项目↔金标准项目（名字相似度≥阈值，平局看属性）；返回 (matched[(pi,gi)], unmatched_pred[idx], unmatched_gold[idx])。"""
    candidates = []
    for pi, p in enumerate(pred_projects):
        for gi, g in enumerate(gold_projects):
            sim = _name_sim(p.get("project_name"), g.get("project_name"))
            if sim >= threshold:
                candidates.append((sim, _attr_bonus(p, g), pi, gi))
    candidates.sort(key=lambda c: (c[0], c[1]), reverse=True)
    used_p, used_g, matched = set(), set(), []
    for _sim, _bonus, pi, gi in candidates:
        if pi in used_p or gi in used_g:
            continue
        used_p.add(pi)
        used_g.add(gi)
        matched.append((pi, gi))
    unmatched_pred = [i for i in range(len(pred_projects)) if i not in used_p]
    unmatched_gold = [i for i in range(len(gold_projects)) if i not in used_g]
    return matched, unmatched_pred, unmatched_gold
```

```python
def score_entities(pred_entities, gold_entities):
    """比较两个项目的中企集合：返回 (name_p, name_r, name_f1, role_correct, role_matched)。
    name 按归一名集合算 P/R/F1；role 仅对两边同名企业比 involvement_type。"""
    pmap = {_norm_name(e.get("name")): e.get("involvement_type") for e in pred_entities if e.get("name")}
    gmap = {_norm_name(e.get("name")): e.get("involvement_type") for e in gold_entities if e.get("name")}
    ps, gs = set(pmap), set(gmap)
    if not ps and not gs:
        return (1.0, 1.0, 1.0, 0, 0)
    inter = ps & gs
    name_p = len(inter) / len(ps) if ps else 0.0
    name_r = len(inter) / len(gs) if gs else 0.0
    name_f1 = 2 * name_p * name_r / (name_p + name_r) if (name_p + name_r) else 0.0
    role_matched = len(inter)
    role_correct = sum(1 for n in inter if cmp_enum("involvement_type", pmap[n], gmap[n]))
    return (name_p, name_r, name_f1, role_correct, role_matched)
```

```python
PROJECT_ENUM_FIELDS = ("category", "project_stage", "financing_model")
PROJECT_NUMERIC_FIELDS = ("capacity_mw", "total_investment_usd")

def score_article(pred: dict, gold: dict) -> dict:
    """单篇文章打分：status 闸门 → 贪心对齐 → 对上项目逐字段/实体打分。
    返回 {status_correct, scored, project_set:(matched,pred_n,gold_n)|None,
          field_hits:{field:(correct,total)}, entity:(sum_name_f1,n_matched,role_correct,role_matched)|None}。"""
    out = {"status_correct": cmp_status(pred.get("status"), gold.get("status")),
           "scored": False, "project_set": None, "field_hits": {}, "entity": None}
    if not out["status_correct"] or gold.get("status") != "valid":
        return out
    out["scored"] = True
    pred_projects = pred.get("projects") or []
    gold_projects = gold.get("projects") or []
    matched, _unp, _ung = align_projects(pred_projects, gold_projects)
    out["project_set"] = (len(matched), len(pred_projects), len(gold_projects))
    all_fields = PROJECT_ENUM_FIELDS + PROJECT_NUMERIC_FIELDS
    field_correct = {f: 0 for f in all_fields}
    field_total = {f: 0 for f in all_fields}
    sum_name_f1 = 0.0
    role_correct = role_matched = 0
    for pi, gi in matched:
        p, g = pred_projects[pi], gold_projects[gi]
        for f in PROJECT_ENUM_FIELDS:
            field_total[f] += 1
            if cmp_enum(f, p.get(f), g.get(f)):
                field_correct[f] += 1
        for f in PROJECT_NUMERIC_FIELDS:
            field_total[f] += 1
            if cmp_numeric(p.get(f), g.get(f)):
                field_correct[f] += 1
        _np, _nr, nf1, rc, rm = score_entities(p.get("entities") or [], g.get("entities") or [])
        sum_name_f1 += nf1
        role_correct += rc
        role_matched += rm
    out["field_hits"] = {f: (field_correct[f], field_total[f]) for f in all_fields}
    out["entity"] = (sum_name_f1, len(matched), role_correct, role_matched)
    return out
```

```python
"""flat_prediction：把 s01b 原始提取结果归一成文章级 {status, projects[]}，每项目含字段 + entities[{name,involvement_type}]。"""

_STATUS_MAP = {"valid_project": "valid", "no_chinese_participation": "no_chinese", "irrelevant": "irrelevant"}

def _flat_project(proj: dict) -> dict:
    """单个项目 → 扁平 dict（字段 + entities 名字/角色对）。"""
    common = proj.get("common_info", {}) or {}
    entities = []
    for e in proj.get("chinese_entities", []) or []:
        name = e.get("Implementing Chinese Entity", "")
        if name:
            entities.append({"name": name, "involvement_type": e.get("Type of involvement of Chinese Entities", "")})
    return {
        "project_name": common.get("Project Name", ""),
        "category": common.get("Category", ""),
        "capacity_mw": common.get("Capacity (MW)", ""),
        "project_stage": common.get("Project Stage", ""),
        "location": common.get("Location(State)", ""),
        "total_investment_usd": common.get("Total Project Investment (Million USD)", ""),
        "financing_model": common.get("Financing Model", ""),
        "entities": entities,
    }

def flat_prediction(result: dict) -> dict:
    """原始结果 → {status(归一), projects:[扁平项目,...]}；非 valid 时 projects=[]。"""
    raw_status = result.get("status")
    status = _STATUS_MAP.get(raw_status, raw_status if raw_status is not None else "")
    if status != "valid":
        return {"status": status, "projects": []}
    projects = [_flat_project(p) for p in (result.get("projects") or [])]
    return {"status": status, "projects": projects}
```

```python
"""gold_loader：读取并用 Pydantic 校验金标准评测集（文章级 status + projects[]），失败带行号即报错。"""
import json
import logging
from typing import Literal, Optional

from pydantic import BaseModel, ConfigDict, Field, ValidationError

logger = logging.getLogger(__name__)


class Entity(BaseModel):
    """项目内一家中企：name 必填非空，involvement_type 可空。"""
    model_config = ConfigDict(extra="forbid")
    name: str = Field(min_length=1)
    involvement_type: Optional[str] = None


class Project(BaseModel):
    """金标准单个项目：project_name 仅对齐用，5 个评分字段 + location 可空，entities 列表。"""
    model_config = ConfigDict(extra="forbid")
    project_name: Optional[str] = None
    category: Optional[str] = None
    project_stage: Optional[str] = None
    financing_model: Optional[str] = None
    capacity_mw: Optional[str] = None
    total_investment_usd: Optional[str] = None
    location: Optional[str] = None
    entities: list[Entity] = Field(default_factory=list)


class GoldRecord(BaseModel):
    """金标准一行（一篇文章）：source_link 必填，status 文章级三选一，projects 列表。"""
    model_config = ConfigDict(extra="forbid")
    source_link: str = Field(min_length=1)
    article_id: Optional[str] = None
    status: Literal["valid", "no_chinese", "irrelevant"]
    projects: list[Project] = Field(default_factory=list)


def load(path) -> list[dict]:
    """逐行读取 gold_set.jsonl 并校验，任一行非法立即带行号抛错；返回纯 dict 列表。"""
    records: list[dict] = []
    with open(path, "r", encoding="utf-8") as f:
        for lineno, raw in enumerate(f, start=1):
            line = raw.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError as e:
                logger.error("gold_loader 第 %d 行 JSON 解析失败: %s | 内容: %s", lineno, e, line)
                raise ValueError(f"gold_set 第 {lineno} 行不是合法 JSON: {e} | 内容: {line}") from e
            try:
                rec = GoldRecord.model_validate(obj)
            except ValidationError as e:
                logger.error("gold_loader 第 %d 行校验失败: %s | 内容: %s", lineno, e, line)
                raise ValueError(f"gold_set 第 {lineno} 行校验失败: {e} | 内容: {line}") from e
            records.append(rec.model_dump())
    logger.info("gold_loader 已加载 %d 条金标准记录", len(records))
    return records
```

```python
"""csv_to_gold：把人工标注后的多行 draft.csv 按 source_link 聚合成 gold_set.jsonl（correct-only 约定）。

约定：<field> 留空 = 沿用 ai_<field>；填了 = 覆盖；写空标记（- 等）= 置空。
entities 单元格写 名字(角色); 名字(角色)，转成 [{name,involvement_type}]。
status 取该篇各行的非空值；无 status 的文章跳过并告警。
"""
import argparse
import json
import logging
import pathlib
import re
import sys

import pandas as pd
from pydantic import BaseModel, field_validator

_root = next(p for p in pathlib.Path(__file__).resolve().parents if (p / "bri_enums.py").exists())
if str(_root) not in sys.path:
    sys.path.insert(0, str(_root))

from eval import gold_loader  # noqa: E402

logger = logging.getLogger(__name__)

_EMPTY_MARKERS = {"-", "—", "空", "<空>", "none", "na", "n/a", "nan"}
_VALUE_FIELDS = ("category", "capacity_mw", "project_stage", "location",
                 "total_investment_usd", "financing_model")
_ENTITY_RE = re.compile(r"^\s*(?P<name>[^()（）]+?)\s*(?:[（(](?P<role>[^()（）]*)[)）])?\s*$")


class ConvertArgs(BaseModel):
    """入口参数校验：输入 CSV 必须存在、输出父目录必须存在。"""
    csv_path: pathlib.Path
    out_path: pathlib.Path

    @field_validator("csv_path")
    @classmethod
    def _csv_exists(cls, v):
        if not v.is_file():
            raise ValueError(f"输入 CSV 不存在: {v}")
        return v

    @field_validator("out_path")
    @classmethod
    def _out_parent(cls, v):
        if not v.parent.is_dir():
            raise ValueError(f"输出目录不存在: {v.parent}")
        return v


def _chosen(row, field):
    """字段最终值：<field> 填了用它，否则沿用 ai_<field>。"""
    corrected = str(row.get(field, "")).strip()
    return corrected if corrected else str(row.get(f"ai_{field}", "")).strip()


def _norm_value(raw):
    """单值字段：空标记/空串 → None，否则原串。"""
    return None if (not raw or raw.lower() in _EMPTY_MARKERS) else raw


def _parse_entities(raw):
    """把 名字(角色); 名字(角色) 解析成 [{name,involvement_type}]；空标记/空 → []。"""
    if not raw or raw.lower() in _EMPTY_MARKERS:
        return []
    out = []
    for part in raw.replace("；", ";").split(";"):
        part = part.strip()
        if not part:
            continue
        m = _ENTITY_RE.match(part)
        if not m or not m.group("name").strip():
            raise ValueError(f"实体格式无法解析（应为 名字(角色)）: {part!r}")
        role = (m.group("role") or "").strip()
        out.append({"name": m.group("name").strip(), "involvement_type": role or None})
    return out


def _row_to_project(row):
    """一行（有项目名/字段）→ 一个 project dict。"""
    proj = {"project_name": _chosen(row, "project_name") or None,
            "entities": _parse_entities(_chosen(row, "entities"))}
    for f in _VALUE_FIELDS:
        proj[f] = _norm_value(_chosen(row, f))
    return proj


def _has_project(row):
    """该行是否承载一个项目（有项目名或任一项目字段或实体）。"""
    if _chosen(row, "project_name"):
        return True
    if _parse_entities(_chosen(row, "entities")):
        return True
    return any(_norm_value(_chosen(row, f)) for f in _VALUE_FIELDS)


def convert(csv_path: pathlib.Path, out_path: pathlib.Path) -> int:
    """读多行标注 CSV → 按 source_link 聚合 → 写 gold_set.jsonl，返回文章数；末尾 gold_loader 校验。"""
    try:
        df = pd.read_csv(csv_path, dtype=str, keep_default_na=False, encoding="utf-8-sig")
    except (OSError, pd.errors.ParserError) as e:
        logger.error("读取标注 CSV 失败: %s | %s", csv_path, e)
        raise
    records, skipped = [], 0
    for source_link, group in df.groupby("source_link", sort=False):
        sl = str(source_link).strip()
        if not sl:
            continue
        status = ""
        for _, row in group.iterrows():
            status = _chosen(row, "status")
            if status:
                break
        if not status:
            skipped += 1
            logger.warning("文章 %s 未标注 status，跳过", sl)
            continue
        projects = [_row_to_project(row) for _, row in group.iterrows() if _has_project(row)]
        article_id = ""
        for _, row in group.iterrows():
            article_id = str(row.get("article_id", "")).strip()
            if article_id:
                break
        rec = {"source_link": sl, "status": status, "projects": projects}
        if article_id:
            rec["article_id"] = article_id
        records.append(rec)
    try:
        with open(out_path, "w", encoding="utf-8") as f:
            for rec in records:
                f.write(json.dumps(rec, ensure_ascii=False) + "\n")
    except OSError as e:
        logger.error("写 gold_set.jsonl 失败: %s | %s", out_path, e)
        raise
    gold_loader.load(out_path)
    logger.info("已写出 %d 篇 gold（跳过未标注 %d 篇）→ %s", len(records), skipped, out_path)
    return len(records)


def _setup_logging():
```
