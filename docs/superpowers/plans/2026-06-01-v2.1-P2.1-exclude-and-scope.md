# v2.1 P2.1 — exclude 机制 + 录入范围对齐 + critic 枚举约束 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把"该不该在库里"做成一等公民的可逆 `excluded` 标记（提取先标、v4-pro review 确认/分流），统一提取/review 三层的"录入范围"定义（发电[含火电]+新能源制造业+输变电），并约束 critic 只产合法枚举值。

**Architecture:** 改动分两侧。**提取侧**（s01b staged prompts + 写库）：S1 范围扩到三类、S2 收紧「综合能源」并新增 `in_scope` 出口、写库时 `in_scope=false → excluded=1`。**review 侧**（`review/`）：critic prompt 对齐范围 + 注入合法枚举、`Verdict` 新增 `exclude` 值、`apply` 新增 exclude 分流（conf≥1.0 自动置 excluded、conf<1.0 入队 `exclude_suggested`）、顺带修 capacity 写解析值 / project_name 转人工。**传播**：s02a 去重时把成员 `excluded` 以"任一成员排除即排除"的 OR 语义传播到 `canonical_projects.excluded`。枚举单一可信源 `enums_mapping.json`：火电 `excluded:true→false`。

**Tech Stack:** Python 3.12 + sqlite3 + pydantic + pytest；`uv` 管理 venv。枚举走 `bri_enums`，数值走 `bri_parse`。LLM 在测试中一律 mock（零成本）。

```python
    sys.path.insert(0, str(_root))

logger = logging.getLogger(__name__)


class Verdict(BaseModel):
    """单字段核验结论：verdict 四选一（含 exclude），confidence∈[0,1]。"""
    field: str
    verdict: Literal["ok", "wrong", "unsupported", "exclude"]
    suggested_value: Optional[str] = None
    confidence: float = Field(ge=0.0, le=1.0)
    evidence: str = ""


def _build_critic_prompt(*, article: dict, project: dict, suspicious: list) -> str:
    """拼 critic prompt：注入合法 category/involvement 枚举 + 三类录入范围 + exclude 通道。"""
    cat_list = ", ".join(c for c in bri_enums.valid_categories() if c != "Unknown")
    inv_list = ", ".join(bri_enums.valid_involvement())
    return f"""你是严谨的核查员。下面是一篇新闻原文和从中提取的某个项目记录。
请逐字段核验提取是否被原文支持，并额外挑战：这真的是「在录入范围内、且有中国企业参与」的项目吗？

### 录入范围（只认这三类）
① 发电项目（任何燃料，含火电/燃气/煤电/核电/光伏/风电/水电/储能/生物质/地热/综合能源）；
② 新能源制造业（以产能计的组件厂/电池厂/多晶硅厂等工厂）；
③ 输变电/电网工程。
不在范围（应 exclude）：油气开采/炼化/管道/LNG 储运、采矿冶炼、纯民用基建（学校/医院/机场/铁路/港口/灌溉/供水/数据中心/住宅/工业园区本身）、纯运维检修、纯框架 MOU、纯评论榜单、纯领导访问。
注意：范围措辞用「电力/发电（含火电）+ 制造业 + 输变电」，**不要**用"可再生能源"这种窄词去否定火电。

原文标题：{article.get('title', '')}
原文正文：{article.get('content', '')}

提取记录（JSON）：{json.dumps(project, ensure_ascii=False)}
重点核查这些可疑字段：{suspicious}

### 合法枚举（字段纠正只能从这里选；选不出就别猜）
- category: {cat_list}
- involvement_type: {inv_list}

### 输出（严格 JSON）
{{"verdicts":[{{"field":"字段名或 status","verdict":"ok|wrong|unsupported|exclude","suggested_value":"建议值或null","confidence":0.0,"evidence":"原文依据"}}]}}
- verdict=ok：原文支持当前值。
- verdict=wrong：字段值错，给 suggested_value。**category/involvement 的 suggested_value 必须是上面合法枚举之一**；若合法枚举里没有合适值（如原文是油气/天然气），**不要**硬填——改用 exclude。
- verdict=unsupported：原文找不到依据。
- verdict=exclude：该项目**不在录入范围**或**无中企参与**（field 写 "status"）。判 exclude 时**不要**借 category/involvement 表达。证据充分、确定该排除时 confidence 给 1.0；存疑给 <1.0（会走人工抽检）。
"""


def review_project(*, article: dict, project: dict, suspicious: list, llm) -> list:
    """对一个可疑项目跑批判核验；返回 Verdict 列表；LLM/解析失败由 llm 层抛出，输出非法则抛 ValidationError。"""
    prompt = _build_critic_prompt(article=article, project=project, suspicious=suspicious)
    raw = llm.complete(prompt, stage="critic", json_mode=True)
    if not isinstance(raw, dict):
        logger.warning("critic LLM 返回非 dict: type=%s | raw=%r", type(raw).__name__, raw)
        items = []
    else:
        items = raw.get("verdicts") or []
        if not isinstance(items, list):
            logger.warning("critic LLM verdicts 非 list: type=%s", type(items).__name__)
            items = []
    try:
        return [Verdict.model_validate(it) for it in items]
    except ValidationError as e:
        logger.error("critic 输出校验失败: %s | 原始=%s", e, raw)
        raise
```

```python
"""apply：按 verdict 分流——字段值高置信纠正自动改库；status/存在性挑战与低置信入 review_queue；exclude 判决高置信自动置 excluded（可逆），低置信入 exclude_suggested 清单。"""
import logging
from typing import Callable, Optional

from review import queue

logger = logging.getLogger(__name__)


def route(*, canonical_id, source_link, verdicts, review_db, update_field,
          conf_threshold: float = 0.8, value_validator: Optional[Callable] = None,
          exclude_fn: Optional[Callable] = None, exclude_auto_threshold: float = 1.0) -> dict:
    """处理一个项目的 verdict 列表。

    - update_field(canonical_id, field, new_value)：字段纠正闭包（事务改库），调用方注入。
    - value_validator(field, suggested_value)->bool：自动改前校验建议值合法性（None=不校验）。
    - exclude_fn(canonical_id, reason)：exclude 自动置位闭包（事务改库 excluded=1），调用方注入；None 时退化为入队。
    - exclude_auto_threshold：exclude 自动置位的最低 confidence（默认 1.0，即仅"确定"才自动排除）。
    返回 {auto_corrected, auto_excluded, queued, reasons}。"""
    auto_corrected = auto_excluded = queued = 0
    reasons: list[str] = []
    for v in verdicts:
        if v.verdict == "ok":
            continue
        # exclude 判决：高置信 + 注入了 exclude_fn → 自动置 excluded（可逆、记日志）；否则入人工清单
        if v.verdict == "exclude":
            if exclude_fn is not None and v.confidence >= exclude_auto_threshold:
                exclude_fn(canonical_id, v.evidence)
                auto_excluded += 1
            else:
                queue.add_item(review_db, canonical_id=canonical_id, source_link=source_link,
                               reason_type="exclude_suggested", field="status",
                               current_value=None, suggested_value="exclude",
                               critic_confidence=v.confidence, evidence=v.evidence)
                queued += 1
                reasons.append("exclude_suggested")
            continue
        # status / 存在性挑战：高风险，绝不自动改，一律入清单
        if v.field == "status":
            queue.add_item(review_db, canonical_id=canonical_id, source_link=source_link,
                           reason_type="status_challenge", field="status",
                           current_value=None, suggested_value=v.suggested_value,
                           critic_confidence=v.confidence, evidence=v.evidence)
            queued += 1
            reasons.append("status_challenge")
            continue
        # 字段值纠正：高置信 + 有建议值 + 建议值通过合法性校验 → 自动改；否则入清单
        value_ok = value_validator is None or value_validator(v.field, v.suggested_value)
        if v.verdict == "wrong" and v.suggested_value and v.confidence >= conf_threshold and value_ok:
            update_field(canonical_id, v.field, v.suggested_value)
            auto_corrected += 1
        else:
            queue.add_item(review_db, canonical_id=canonical_id, source_link=source_link,
                           reason_type="low_confidence_field", field=v.field,
                           current_value=None, suggested_value=v.suggested_value,
                           critic_confidence=v.confidence, evidence=v.evidence)
            queued += 1
            reasons.append("low_confidence_field")
    return {"auto_corrected": auto_corrected, "auto_excluded": auto_excluded,
            "queued": queued, "reasons": reasons}
```

```python
def _make_exclude_field(rid, wconn, _dry_run, _would_change, _failed_counter):
    """工厂：为每条记录生成 exclude_field 闭包（事务置 excluded=1, exclude_source='review'）。"""
    def exclude_field(canonical_id, reason):
        """执行（或模拟）置 excluded；dry_run 时只记 would_change。"""
        if _dry_run:
            logger.info("[DRY_RUN] 会将 id=%s 置 excluded=1（review），理由=%r", rid, reason)
            _would_change.append({"id": rid, "field": "excluded", "new_value": 1, "reason": reason})
            return
        try:
            with wconn:
                wconn.execute(
                    "UPDATE extracted_projects SET excluded=1, exclude_source='review', exclude_reason=? WHERE id=?",
                    (reason, rid),
                )
            logger.info("已置 excluded id=%d source=review 理由=%r", rid, reason)
        except sqlite3.Error as e:
            logger.error("置 excluded 失败 id=%d: %s", rid, e)
            _failed_counter[0] += 1
    return exclude_field
```
